from typing import Optional
import pandas as pd

from helpers.io import load_config
from helpers.yaml import load_yaml

def b6_operation(property: 'BaseProperty', pathway: 'BasePathway', start_year: Optional[int], end_year: Optional[int]) -> pd.Series:
    years = list(range(start_year, end_year + 1))

    from models.property import Property
    from models.pathway import BasePathway, CustomPathway, BaselinePathway

    if isinstance(pathway, CustomPathway):
        values = [14 for year in years]
    else:
        if property.initial_state == "BUILDING":
            values = [4 for year in years]
        else:
            values = [0 for year in years]
    
    return pd.Series(data=values, index=years, name="b6_operation")

def load_emission_factors():
    config = load_config()
    energy_factors_path = config['reference_data']['energy']
    emission_factors = load_yaml(energy_factors_path)

    return emission_factors['energy']

def calculate_operational_carbon(usage, year, verbose=False):
    emission_factors = load_emission_factors()  
    total_carbon_footprint = 0

    for energy_source, amount in usage.items():
        if energy_source in emission_factors:
            factor_data = emission_factors[energy_source]

            if isinstance(factor_data, dict):
                years = list(map(int, factor_data.keys()))
                factors = list(map(float, factor_data.values()))
                from helpers.math import interpolate_values
                interpolated_factor = interpolate_values(year, years, factors)
            else:
                interpolated_factor = factor_data

            carbon_footprint = amount * interpolated_factor / 1000  
            total_carbon_footprint += carbon_footprint

            if verbose:
                print(f"{energy_source}: Usage = {amount}, Year = {year}, Factor = {interpolated_factor}, Carbon = {carbon_footprint} kgCO2")
        else:
            if verbose:
                print(f"No emission factors found for {energy_source}")

    if verbose:
        print(f"Total Carbon Footprint for year {year}: {total_carbon_footprint} kgCO2")
    return total_carbon_footprint
